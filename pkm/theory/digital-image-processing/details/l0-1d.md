import numpy as np
import matplotlib.pyplot as plt


# 设置字体以支持中文显示
plt.rcParams['font.sans-serif'] = ['Arial Unicode MS', 'SimHei', 'Heiti TC', 'Microsoft YaHei']
plt.rcParams['axes.unicode_minus'] = False # 解决负号显示问题


def l0_filter_1d(g, lambda_val=0.01, max_iter=1000, tol=1e-6):
    """
    一维 L0 滤波器：平滑信号并保留指定数量的边缘（通过调整 lambda 间接控制）
    参数:
    g: 输入一维信号 (numpy array)
    lambda_val: 惩罚系数，控制边缘数量（越大，边缘越少；越小，边缘越多）
    max_iter: 最大迭代次数
    tol: 收敛阈值（残差变化小于此值时停止）
    返回:
    f: 滤波后的信号
    """
    n = len(g)
    f = g.copy() # 初始化滤波结果为原始信号
    for iter in range(max_iter):
        f_old = f.copy()
        # 遍历每个像素，计算最优更新值
        for p in range(1, n-1): # 边界像素不更新（保持原信号）
            # 提取左右邻居
            f_left, f_right = f[p-1], f[p+1]
            g_p = g[p]
            # 计算“保留当前值”和“调整为左/右邻居”的残差与惩罚项
            # 情况1：保持当前值 f[p]
            cost_keep = (f[p] - g_p)**2 + lambda_val * (int(f[p] != f_left) + int(f[p] != f_right))
            # 情况2：调整为左邻居值 f_left
            cost_left = (f_left - g_p)**2 + lambda_val * (int(f_left != f[p-2]) if p > 1 else 0 + int(f_left != f_right))
            # 情况3：调整为右邻居值 f_right
            cost_right = (f_right - g_p)**2 + lambda_val * (int(f_right != f_left) + int(f_right != f[p+2]) if p < n-2 else 0)
            # 选择成本最小的方案
            costs = [cost_keep, cost_left, cost_right]
            best_idx = np.argmin(costs)
            if best_idx == 0:
                f[p] = f[p] # 保持原值
            elif best_idx == 1:
                f[p] = f_left # 变为左邻居值
            else:
                f[p] = f_right # 变为右邻居值
        # 检查收敛（残差变化是否小于阈值）
        diff = np.sum((f - f_old)**2)
        if diff < tol:
            print(f"收敛于第 {iter+1} 次迭代")
            break
    return f

def count_edges(f):
    """计算一维信号的边缘数量（相邻像素变化的次数）"""
    return np.sum(np.abs(np.diff(f)) > 1e-6) # 避免浮点数精度问题

# ------------------- 测试代码 -------------------
if __name__ == "__main__":
    # 生成测试信号：带噪声的阶梯信号（模拟边缘）
    np.random.seed(42)
    n = 2000
    g = np.zeros(n)
    # 构造3个边缘（阶梯）
    g[:500] = 5
    g[500:1000] = 3
    g[1000:1500] = 10
    g[1500:] = 2
    # 添加噪声
    noise = np.random.normal(0, 0.5, n)
    g += noise
    # 测试不同的 lambda（控制边缘数量）
    lambda_list = [1.0, 5.0, 10.0] # lambda 越大，边缘越少
    fig, axes = plt.subplots(len(lambda_list)+1, 1, figsize=(10, 12))
    # 绘制原始信号
    axes[0].plot(g, color='gray', alpha=0.6, label='原始信号（带噪声）')
    axes[0].set_title('原始信号')
    axes[0].legend()
    # 对每个 lambda 进行滤波并绘图
    for i, lam in enumerate(lambda_list):
        f = l0_filter_1d(g, lambda_val=lam)
        edge_num = count_edges(f)
        axes[i+1].plot(g, color='gray', alpha=0.4, label='原始信号')
        axes[i+1].plot(f, color='red', label=f'滤波后（λ={lam}, 边缘数={edge_num}）')
        axes[i+1].set_title(f'L0 滤波结果（λ={lam}）')
        axes[i+1].legend()
    plt.tight_layout()
    plt.show()

```java
package pro_money;  
  
import java.text.SimpleDateFormat;  
import java.util.Date;  
import java.util.Scanner;  
  
class History{  
    private double money;  
    private Date date;  
    private double balance;  
    private String info;  
  
    public History(double money, Date date, double balance, String info) {  
        this.money = money;  
        this.date = date;  
        this.balance = balance;  
        this.info = info;  
    }  
    @Override  
    public String toString(){  
        SimpleDateFormat sdf = new SimpleDateFormat("yyyy-MM-dd HH:mm");  
        if(money > 0)  
            return "收入\t+" + money + "\t" + sdf.format(date) + "\t" + balance;  
        else  
            return info + "\t" + money + "\t" + sdf.format(date) + "\t" + balance;  
    }}  
  
class SmallChangeApp {  
    private Scanner scanner = new Scanner(System.in);  
    private History[] historys = new History[64];  
    private int length = 0;  
    private double balance = 0.0;  
    private double temp = 0.0;  
  
    public void showMenu(){  
        System.out.println("===========零钱通===========");  
        System.out.println("1 零钱通明细");  
        System.out.println("2 收益入账");  
        System.out.println("3 消费");  
        System.out.println("4 退出");  
        System.out.print("请输入您的选择:");  
    }  
    public void showInfo(){  
        System.out.println("-----------明细-----------");  
        for(int i=0; i<length; i++)  
            System.out.println(historys[i].toString());  
    }  
    public void addIncome(){  
        System.out.print("收入金额:");  
        temp = scanner.nextDouble();  
        if(temp <= 0){  
            System.out.println("金额需大于零");  
            return;  
        }        balance += temp;  
        historys[length++] = new History(temp, new Date(), balance, "");  
    }  
    public void addPay(){  
        System.out.print("消费金额:");  
        temp = scanner.nextDouble();  
        if(temp <= 0){  
            System.out.println("金额需大于零");  
            return;  
        }        if(temp > balance){  
            System.out.println("余额不足");  
            return;  
        }        balance -= temp;  
        System.out.print("消费类型:");  
        historys[length++] = new History(-temp, new Date(),balance, scanner.next());  
    }  
    public void run(){  
        while(true){  
            showMenu();  
            switch(scanner.nextInt()){  
                case 1:  
                    showInfo();  
                    break;  
                case 2:  
                    addIncome();  
                    break;  
                case 3:  
                    addPay();  
                    break;  
                case 4:  
                    return;  
                default:  
                    System.out.println("错误选择");  
                    break;  
            }        }    }}  
  
public class SmallChangeSys {  
    public static void main(String[] args){  
        SmallChangeApp app = new SmallChangeApp();  
        app.run();  
    }}
```

```java
package ex_chapter8;  
  
class Person{  
    String name;  
    int age;  
    String job;  
    public Person(String name, int age, String job) {  
        this.name = name;  
        this.age = age;  
        this.job = job;

public class Tester01{  
    public static void main(String[] args){  
        Person[] ps = new Person[3];  
        ps[0] = new Person("Jack", 20, "Student");  
        ps[1] = new Person("Tom", 24, "Cow Horse");  
        ps[2] = new Person("Jassica", 22, "Doctor");  
        ps = sort(ps);  
        for(int i=0; i<ps.length; i++)  
            System.out.println(ps[i].age);  
    }  
    public static Person[] sort(Person[] a){  
        Person temp;  
        if(a.length <= 1)  
            return a;  
        for(int i=0; i<a.length-1; i++){  
            for(int j=i; j<a.length-1; j++){  
                if(a[j].age<a[j+1].age){  
                    temp = a[j];  
                    a[j] = a[j+1];  
                    a[j+1] = temp;  
                }            
            }        
        }        
        return a;  
    }
```

```java
package ex_chapter8;  
  
public class Test08 {  
    public static void main(String[] args){  
        CheckingAccount c = new CheckingAccount(1000);  
        c.deposit(10);  
        c.withdraw(9);  
        System.out.println(c.getBalance()); // 999.0  
  
        SavingAccount s = new SavingAccount(1000);  
        s.deposit(100);  
        s.deposit(100);  
        s.deposit(100);  
        System.out.println(s.getBalance()); // 1300  
        s.deposit(100);  
        System.out.println(s.getBalance()); // 1399  
        s.earnMonthlyInterest();  
        System.out.println(s.getBalance()); // 1412.99  
    }  
}  
  
class BankAccount{  
    private double balance;  
    public BankAccount(double initialBalance) {  
        this.balance = initialBalance;  
    }    public void deposit(double amount) {  
        balance += amount;  
    }    public void withdraw(double amount) {  
        balance -= amount;  
    }    //set 和 getBalance方法  
    public void setBalance(double balance){  
        this.balance = balance;  
    }  
    public double getBalance(){  
        return this.balance;  
    }}  
  
class CheckingAccount extends BankAccount{// 利率  
    public CheckingAccount(double initialBalance) {  
        super(initialBalance);  
    }  
    @Override  
    public void deposit(double amount) {  
        super.deposit(amount-1);  
    }    @Override  
    public void withdraw(double amount) {  
        super.withdraw(amount+1);  
    }}  
  
class SavingAccount extends BankAccount{  
    private int count = 3;  
    private double rate = 0.01;  
  
    public SavingAccount(double initialBalance) {  
        super(initialBalance);  
    }  
    public void earnMonthlyInterest(){  
        // 每个月统计利息，并且重置 count        count = 3;  
        super.deposit(getBalance() * rate);  
    }  
    @Override  
    public void deposit(double amount){  
        if(count-- > 0)  
            super.deposit(amount);  
        else  
            super.deposit(amount-1);  
    }  
    @Override  
    public void withdraw(double amount){  
        if(count-- > 0)  
            super.withdraw(amount);  
        else  
            super.withdraw(amount+1);  
    }}
```

程序阅读题在 main 方法中执行：`C c = new C();` 输出什么内容？()

```java
class A { // 超类
    public A() {
        System.out.println("我是A类"); // 3
    }
}

class B extends A { // 父类
    public B() {
        System.out.println("我是B类的无参构造");
    }
    
    public B(String name) {
        // 3
        System.out.println(name + "我是B类的有参构造"); // 2
    }
}

class C extends B { // 子类
    public C() {
        this("hello");
        System.out.println("我是c类的无参构造"); // 0
    }
    
    public C(String name) {
        super("hahah"); // 2
        System.out.println("我是c类的有参构造"); // 1
    }
}
```

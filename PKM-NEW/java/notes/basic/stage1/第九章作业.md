// HouseView.java
package pro_house.view;  
  
import pro_house.domain.House;  
import pro_house.service.HouseService;  
import pro_house.utils.Utility;  
  
/**  
 * 显示界面(界面)  
 * 接受用户输入  
 * 调用其他类完成房屋信息操作  
 */  
public class HouseView{  
    private final HouseService houseService = new HouseService(4);  
    private boolean loop = true;  
    public void mainMenu(){  
        while(loop) {  
            System.out.println("========房屋出租系统菜单========");  
            System.out.println("\t 1 新增房源");  
            System.out.println("\t 2 查找房屋");  
            System.out.println("\t 3 删除房屋");  
            System.out.println("\t 4 修改信息");  
            System.out.println("\t 5 显示房屋");  
            System.out.println("\t 6 退出程序");  
            switch (Utility.readChar()) {  
                case '1':  
                    addHouses();  
                    break;  
                case '2':  
                    searchHouses();  
                    break;  
                case '3':  
                    delHouses();  
                    break;  
                case '4':  
                    modifyHouses();  
                    break;  
                case '5':  
                    listHouses();  
                    break;  
                case '6':  
                    exitHouses();  
                    break;  
                default:  
                    System.out.println("需要输入 1-6 的数字");  
                    break;  
            }        
        }    
    }  
    public void addHouses(){  
        System.out.println("--------新增房源--------");  
        if(houseService.isFull()){  
            System.out.println("房源已满");  
            return;  
        }        
        System.out.print("姓名: ");  
        String name = Utility.readString(8);  
        System.out.print("电话: ");  
        String tel = Utility.readString(12);  
        System.out.print("地址: ");  
        String adder = Utility.readString(16);  
        System.out.print("月租: ");  
        int rent = Utility.readInt();  
        houseService.addHouse(name, tel, adder, rent);  
    }  
    public void delHouses(){  
        System.out.println("--------删除房源--------");  
        if(houseService.isEmpty()){  
            System.out.println("房源已空");  
            return;  
        }        
        House[] houses = houseService.listHouses();  
        for(int i=0; i<houseService.getLength(); i++)  
            System.out.println(houses[i]);  
        System.out.print("请输入房源编号(输入 -1 退出): ");  
        int choice = Utility.readInt();  
        if(choice==-1)  
            return;  
        if(houseService.searchIndexById(choice)==-1){  
            System.out.println("没有这个房源");  
            return;  
        }        
        System.out.print("请确认是否删除(y/n): ");  
        if(Utility.readChar() != 'y'){  
            System.out.println("取消删除");  
            return;  
        }        
        houseService.delHouse(choice);  
    }  
    public void searchHouses(){  
        System.out.println("--------信息查找--------");  
        if(houseService.isEmpty()){  
            System.out.println("目前没有房源, 请先添加");  
            return;  
        }        
        System.out.print("请输入要查找的 id(max=" + houseService.getLength() + "): ");  
        House house = houseService.searchById(Utility.readInt());  
        if(house==null)  
            System.out.println("错误的 id, 查找失败");  
        else{  
            System.out.println("编号\t\t房主\t\t电话\t\t地址\t\t月租\t\t已出租");  
            System.out.println(house);  
        }    
    }  
    public void modifyHouses(){  
        System.out.println("--------修改信息--------");  
        if(houseService.isEmpty()){  
            System.out.println("目前没有房源, 请先添加");  
            return;  
        }        
        System.out.print("请输入要查找的 id(max=" + houseService.getLength() + "): ");  
        int choice = Utility.readInt();  
        House house = houseService.searchById(choice);  
        if(house==null)  
            System.out.println("错误的 id, 查找失败");  
        else{  
            System.out.println("编号\t\t房主\t\t电话\t\t地址\t\t月租\t\t已出租");  
            System.out.println(house);  
            System.out.print("房主: ");  
            String name = Utility.readString(8);  
            name = name.isEmpty() ? house.getName() : name;  
            System.out.print("电话: ");  
            String phone = Utility.readString(12);  
            phone = phone.isEmpty() ? house.getPhone() : phone;  
            System.out.print("地址: ");  
            String addr = Utility.readString(16);  
            addr = addr.isEmpty() ? house.getAddress() : addr;  
            System.out.print("月租: ");  
            String _rent = Utility.readString(8);  
            int rent = _rent.isEmpty() ? house.getRent() : Integer.parseInt(_rent);  
            System.out.print("已出租: ");  
            String _state = Utility.readString(8);  
            boolean state = _state.isEmpty() ? house.getState() : Boolean.parseBoolean(_state);  
            houseService.modifyHouse(choice,name,phone,addr,rent,state);  
        }    
    }  
    public void listHouses(){  
        System.out.println("--------房源信息--------");  
        System.out.println("编号\t\t房主\t\t电话\t\t地址\t\t月租\t\t已出租");  
        House[] houses = houseService.listHouses();  
        for(int i=0; i<houseService.getLength(); i++)
            System.out.println(houses[i]);  
    }  
    public void exitHouses(){  
        System.out.print("请确认是否退出(y/n): ");  
        if(Utility.readChar() != 'y')  
            return;  
        System.out.println("您已退出系统");  
        loop = false;  
    }
}

//HouseServe.java
package pro_house.service;  
  
import pro_house.domain.House;  
import pro_house.utils.Utility;  
  
/**  
 * 业务层  
 * 响应 HouseView 的调用，完成信息的操作(crud )  
 */public class HouseService{  
    private int total=1;  
    private final House[] houses;  
    private int length;  
  
    public int getLength() {  
        return length;  
    }  
    private int getNewId(){  
        boolean[] ids = new boolean[total];  
        for(int i=0; i<length; i++)  
            ids[houses[i].getId()-1] = true;  
        for(int i=0; i<total; i++)  
            if(!ids[i])  
                return i+1;  
        return 1;  
    }  
    public HouseService(int size) {  
        houses = new House[size];  
        total = size;  
        // for testing  
        houses[0] = new House(1, "Jack", "112", "海淀区", 2000, false);  
        length = 1;  
    }  
    public boolean isFull(){  
        return houses.length == length;  
    }  
    public boolean isEmpty(){  
        return length == 0;  
    }  
    public int searchIndexById(int id){  
        for(int i=0; i<length; i++)  
            if(houses[i].getId() == id)  
                return i;  
        return -1;  
    }  
    public House searchById(int id){  
        int index = searchIndexById(id);  
        if(index == -1)  
            return null;  
        else  
            return houses[index];  
    }  
    public void addHouse(String name, String phone, String address, int rent){  
        if(isFull())  
            return;  
        houses[length] = new House(getNewId(), name, phone, address, rent, false);  
        length++;  
    }  
    public void delHouse(int id){  
        if(isEmpty())  
            return;  
        for(int i=searchIndexById(id); i<length && i!=-1; i++)  
            houses[i] = houses[i+1];  
        length--;  
    }  
    public void modifyHouse(int id, String name, String phone, String address, int rent, boolean state){  
        if(isEmpty())  
            return;  
        int i = searchIndexById(id);  
        if(i == -1)  
            return;  
        houses[i].setName(name);  
        houses[i].setPhone(phone);  
        houses[i].setAddress(address);  
        houses[i].setRent(rent);  
        houses[i].setState(state);  
    }  
    public House[] listHouses(){  
        return houses;  
    }
}

//House.java
package pro_house.domain;  
  
/**  
 * 房屋类  
 */  
public class House{  
    private int id; // 编号  
    private String name; // 房主  
    private String phone; // 电话  
    private String address; // 地址  
    private int rent; // 月租  
    private Boolean state; // 状态(true 是已出租)  
  
    public House(int id, String name, String phone, String address, int rent, Boolean state) {  
        this.id = id;  
        this.name = name;  
        this.phone = phone;  
        this.address = address;  
        this.rent = rent;  
        this.state = state;  
    }  
    public int getId() {  
        return id;  
    }  
    public String getName() {  
        return name;  
    }  
    public String getPhone() {  
        return phone;  
    }  
    public String getAddress() {  
        return address;  
    }  
    public int getRent() {  
        return rent;  
    }  
    public Boolean getState() {  
        return state;  
    }  
    public void setId(int id) {  
        this.id = id;  
    }  
    public void setName(String name) {  
        this.name = name;  
    }  
    public void setPhone(String phone) {  
        this.phone = phone;  
    }  
    public void setAddress(String address) {  
        this.address = address;  
    }  
    public void setRent(int rent) {  
        this.rent = rent;  
    }  
    public void setState(Boolean state) {  
        this.state = state;  
    }  
    @Override  
    public String toString() {  
        return id + "\t\t" + name + "\t" + phone + "\t\t" + address + "\t" + rent + "\t" + state;  
    }
}

//Utility.java
package pro_house.utils;  
  
import java.util.Scanner;  
  
/**  
 * 工具类  
 */  
public class Utility {  
    public static char readChar(){  
        Scanner scanner = new Scanner(System.in);  
        return scanner.next().charAt(0);  
    }    
    public static int readInt(){  
        Scanner scanner = new Scanner(System.in);  
        return scanner.nextInt();  
    }    
    public static String readString(int length){  
        Scanner scanner = new Scanner(System.in);  
        String res;  
        while(true){  
            res = scanner.nextLine();  
            if(res.length()<=length) 
                return res;  
            else
                System.out.println("长度需要小于"+length);  
                 
        }    
    }
}
